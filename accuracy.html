import pandas as pd from jiwer import wer # Load Reference & STT Output CSV files ref = pd.read_csv("reference.csv") stt = pd.read_csv("stt_output.csv") df = ref.copy() df["stt_text"] = stt["stt_text"] # Calculate Word Error Rate (WER) and Accuracy (%) df["wer"] = df.apply( lambda r: wer(str(r["reference_text"]), str(r["stt_text"])), axis=1 ) df["accuracy"] = (1.0 - df["wer"]).clip(lower=0) * 100 # Aggregate Average Accuracy per Language summary = df.groupby("language")["accuracy"].mean().reset_index() print(summary)
